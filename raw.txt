local CHO_OF, JUNG_OF, JONG_OF = {}, {}, {}

if #k == 2 then JONG_COMBINE[JONG_OF[k[1]]*100 + k[2]] = i end end
for i = 0, 20 do local k = JUNG_KEYS[i]
if #k == 2 then JUNG_COMBINE[JUNG_OF[k[1]]*1000 + k[2]] = i end end

local function wordToKeys(word)
    local keys = {}
    for _, cp in utf8.codes(word) do
        if cp >= SBASE and cp < SBASE + SCOUNT then
            local s = cp - SBASE
            keys[#keys+1] = CHO[math.floor(s / NCOUNT) + 1]
            for _, k in ipairs(JUNG_KEYS[math.floor((s % NCOUNT) / TCOUNT)]) do keys[#keys+1] = k end
            local t = s % TCOUNT
            if t > 0 then for _, k in ipairs(JONG_KEYS[t]) do keys[#keys+1] = k end end
        else keys[#keys+1] = cp end
    end
    return keys
end

local function render(committed, cho, jung, jong)
    if cho and jung then return committed .. utf8.char(SBASE + (cho*VCOUNT + jung)*TCOUNT + jong)
    elseif cho then return committed .. utf8.char(CHO[cho+1])
    elseif jung then return committed .. utf8.char(JUNG[jung+1]) end
    return committed
end

local function buildTypeStates(word)
    local keys = wordToKeys(word)
    local states, committed = {}, ""
    local cho, jung, jong = nil, nil, 0
    local function commit() committed = render(committed, cho, jung, jong); cho, jung, jong = nil, nil, 0 end
    for _, cp in ipairs(keys) do
        if CHO_OF[cp] then
            if cho and jung then
                if jong == 0 and JONG_OF[cp] then jong = JONG_OF[cp]
                elseif jong > 0 and JONG_COMBINE[jong*100 + cp] then jong = JONG_COMBINE[jong*100 + cp]
                else commit(); cho = CHO_OF[cp] end
            elseif cho then commit(); cho = CHO_OF[cp]
            else cho = CHO_OF[cp] end
        elseif JUNG_OF[cp] then
            if jong > 0 then
                local kt = JONG_KEYS[jong]
                if #kt == 2 then jong = JONG_OF[kt[1]]; commit(); cho = CHO_OF[kt[2]]; jung = JUNG_OF[cp]
                else jong = 0; commit(); cho = CHO_OF[kt[1]]; jung = JUNG_OF[cp] end
            elseif cho and jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            elseif jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            else jung = JUNG_OF[cp] end
        else commit(); committed = committed .. utf8.char(cp) end
        states[#states+1] = render(committed, cho, jung, jong)
    end
    local out = {}
    for _, s in ipairs(states) do if out[#out] ~= s then out[#out+1] = s end end
    if #out == 0 or out[#out] ~= word then out[#out+1] = word end
    return out
end

local function checkIsAttackWord(word)
    local lastChar = ""
    for _, cp in utf8.codes(word) do lastChar = utf8.char(cp) end
    if EndingKillLetters[lastChar] or CustomKillLetters[lastChar] then return true end
    local cp = utf8.codepoint(lastChar)
    if cp and cp >= SBASE and cp < SBASE + SCOUNT then
        local s = cp - SBASE
        local choIdx = math.floor(s / NCOUNT)
        local jongIdx = s % TCOUNT
        if TargetJongseiIndex[jongIdx] or DoubleJongseiIndex[jongIdx] then return true end
        if DoubleChoseongIndex[choIdx] then return true end
    end
    return false
end

local function getPriorityScore(w)
    local len = utf8.len(w)
    local isAttack = checkIsAttackWord(w)
    if isAttack and len >= 5 then return 1
    elseif isAttack then return 2
    elseif len >= 6 then return 3 end
    return 4
end

function TurnCheckManager.getStartingLetter(gameDataInstance)
    if not gameDataInstance then
        local label = findSpectatorLabel()
        if label then
            local matched = string.match(label.Text, "'(.-)'") or string.match(label.Text, '"(.-)"')
            if matched then return matched end
        end
        return ""
    end
    
    local letter = gameDataInstance:GetAttribute("StartingLetter") or gameDataInstance:GetAttribute("Letter")
    if not letter or letter == "" then
        local label = findSpectatorLabel()
        if label then
            letter = string.match(label.Text, "'(.-)'") or string.match(label.Text, '"(.-)"')
        end
    end
    return letter or ""
end

local function getMatchedWordsCount(letter)
    if not letter or letter == "" then return 0 end
    local count = 0
    for _, w in ipairs(WordDB) do
        if string.sub(w, 1, string.len(letter)) == letter then
            local isPass = false
            local len = utf8.len(w)
            local isAttack = checkIsAttackWord(w)
            
            if config.selectedFilterType == "전체" then isPass = true
            elseif config.selectedFilterType == "한방단어" and isAttack then isPass = true
            elseif config.selectedFilterType == "초장문" and len >= 6 then isPass = true
            elseif config.selectedFilterType == "한방 초장문" and (isAttack and len >= 5) then isPass = true
            elseif config.selectedFilterType == "방어단어" and (MainDictMap[w] and not isAttack) then isPass = true end
            
            if isPass then count = count + 1 end
        end
    end
    return count
end

local function sendWordToGame(word)
    if word == "" then return end
    local targetBox = forceLoadGameTextBox()
    if not targetBox then return end
    
    currentTypeSession = currentTypeSession + 1
    local mySession = currentTypeSession
    local targetRealSpeed = tonumber(config.typingSpeed) or 350

    if UserRank == "Premium" and config.antiBanEnabled then
        task.spawn(function()
            task.wait(1.0)
            pcall(function()
                if targetBox then
                    targetBox.Text = ""
                end
            end)
        end)
    end

    pcall(function()
        targetBox.Visible = true
        if targetBox.Parent and targetBox.Parent:IsA("GuiObject") then
            targetBox.Parent.Visible = true
        end
        targetBox:CaptureFocus()

        if targetRealSpeed <= 0 then
            if mySession ~= currentTypeSession then return end
            targetBox.Text = word
        else
            local totalStates = buildTypeStates(word)
            local delayTime = 60 / (targetRealSpeed * 2.8) 
            for _, partialText in ipairs(totalStates) do
                if mySession ~= currentTypeSession then return end
                targetBox.Text = partialText
                task.wait(delayTime)
            end
        end
        if mySession ~= currentTypeSession then return end
        task.wait(0.02)
        
        targetBox:ReleaseFocus(true)
        
        task.spawn(function()
            pcall(function()
                local vim = game:GetService("VirtualInputManager")
                if vim then
                    vim:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
                    task.wait(0.05)
                    vim:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
                end
            end)
        end)
    end)
end

                        for _, playerChairFolder in ipairs(chairsFolder:GetChildren()) do
                            local chaise = playerChairFolder:FindFirstChild("chaise")

                    end

            end
        end

        local rawMain = downloadData(urls.dictUrl)
        if rawMain then
            parseAndInsert(rawMain)
        end

        StatusLabel.Text = "Parsing & Removing Duplicates..."
        
        local rawWordsCount = #tempWords
        local chunkSize = 50 
        local lastYieldTime = os.clock()
        local temporaryMap = {}
        
        for i = 1, rawWordsCount, chunkSize do
            for j = i, math.min(i + chunkSize - 1, rawWordsCount) do
                local word = tempWords[j]
                if not temporaryMap[word] then
                    temporaryMap[word] = true
                    table.insert(WordDB, word)
                    MainDictMap[word] = true
                end
            end
            
            if os.clock() - lastYieldTime > 0.1 then
                task.wait()
                lastYieldTime = os.clock()
            end
        end
        temporaryMap = nil
        
        isDataDownloaded = true
    end)

    task.spawn(function()
        local currentFilled = 0
        for i = 1, totalLength do
            currentFilled = i
            local filledPart = string.rep("|", currentFilled)
            local emptyPart = string.rep(" ", totalLength - currentFilled)
            TextProgressBar.Text = "[" .. filledPart .. emptyPart .. "]"
            StatusLabel.Text = "Injecting Automata Data... " .. math.floor((i / totalLength) * 100) .. "%"
            task.wait(0.03) 
        end
        
        StatusLabel.Text = "Finalizing Graphics Engine..."
        while not isDataDownloaded do
            task.wait(0.02)
        end
        
        task.spawn(function()
            initializeWindUIWindow()
        end)
        
        task.defer(function()
            table.sort(WordDB)
        end)
        
        task.wait(0.1)
        TweenService:Create(MainBackground, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
        TweenService:Create(TitleLabel, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        TweenService:Create(TextProgressBar, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        TweenService:Create(StatusLabel, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        
        task.wait(0.3)
        LoadingScreenGui:Destroy()
        
        collectgarbage("collect")
    end)
end

    if checkWhitelistValid() then
        runLoadingSequence()
    end
