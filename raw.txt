local function wordToKeys(word)
    local keys = {}
    for _, cp in utf8.codes(word) do
        if cp >= SBASE and cp < SBASE + SCOUNT then
            local s = cp - SBASE
            keys[#keys+1] = CHO[math.floor(s / NCOUNT) + 1]
            for _, k in ipairs(JUNG_KEYS[math.floor((s % NCOUNT) / TCOUNT)]) do keys[#keys+1] = k end
            local t = s % TCOUNT
            if t > 0 then for _, k in ipairs(JONG_KEYS[t]) do keys[#keys+1] = k end end
        else keys[#keys+1] = cp end
    end
    return keys
end

local function render(committed, cho, jung, jong)
    if cho and jung then return committed .. utf8.char(SBASE + (cho*VCOUNT + jung)*TCOUNT + jong)
    elseif cho then return committed .. utf8.char(CHO[cho+1])
    elseif jung then return committed .. utf8.char(JUNG[jung+1]) end
    return committed
end

local function buildTypeStates(word)
    local keys = wordToKeys(word)
    local states, committed = {}, ""
    local cho, jung, jong = nil, nil, 0
    local function commit() committed = render(committed, cho, jung, jong); cho, jung, jong = nil, nil, 0 end
    for _, cp in ipairs(keys) do
        if CHO_OF[cp] then
            if cho and jung then
                if jong == 0 and JONG_OF[cp] then jong = JONG_OF[cp]
                elseif jong > 0 and JONG_COMBINE[jong*100 + cp] then jong = JONG_COMBINE[jong*100 + cp]
                else commit(); cho = CHO_OF[cp] end
            elseif cho then commit(); cho = CHO_OF[cp]
            else cho = CHO_OF[cp] end
        elseif JUNG_OF[cp] then
            if jong > 0 then
                local kt = JONG_KEYS[jong]
                if #kt == 2 then jong = JONG_OF[kt[1]]; commit(); cho = CHO_OF[kt[2]]; jung = JUNG_OF[cp]
                else jong = 0; commit(); cho = CHO_OF[kt[1]]; jung = JUNG_OF[cp] end
            elseif cho and jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            elseif jung then
                local comb = JUNG_COMBINE[jung*1000 + cp]
                if comb then jung = comb else commit(); jung = JUNG_OF[cp] end
            else jung = JUNG_OF[cp] end
        else commit(); committed = committed .. utf8.char(cp) end
        states[#states+1] = render(committed, cho, jung, jong)
    end
    local out = {}
    for _, s in ipairs(states) do if out[#out] ~= s then out[#out+1] = s end end
    if #out == 0 or out[#out] ~= word then out[#out+1] = word end
    return out
end

local function checkIsAttackWord(w)
    local lastChar = ""
    for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
    
    local cp = utf8.codepoint(lastChar)
    if not cp or cp < SBASE or cp >= SBASE + SCOUNT then return false end
    
    local s = cp - SBASE
    local jongIdx = s % TCOUNT
    
    local validJong = {
        [16] = true, -- ㅁ
        [25] = true, -- ㅌ
        [24] = true, -- ㅊ
        [23] = true, -- ㅋ
        [27] = true, -- ㅎ
        [22] = true  -- ㅈ
    }
    
    if validJong[jongIdx] then
        return true
    end
    
    return false
end

local function checkIsPTier1AttackWord(w)
    local lastChar = ""
    for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
    if P_TIER1[lastChar] then

end

local function getPriorityScore(w)
    local len = utf8.len(w)
    local isAttack = checkIsAttackWord(w)
    local isSuperLong = (len >= 6)
    local isSuperLongAttack = (isAttack and len >= 4)
    
    local lastChar = ""
    for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
    local isLoopAttack = (P_TIER2[lastChar] or P_TIER3[lastChar])

    if isSuperLongAttack then
        return 1
    elseif isAttack then
        return 2
    elseif isLoopAttack then
        return 3
    elseif isSuperLong then
        return 4
    else
        return 5
    end
end

local function isExcludedEndingWord(w)
    local lastChar = ""
    for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
    
    if lastChar == "발" or lastChar == "레" or lastChar == "뼈" or lastChar == "산" then
        return true
    end
    
    local cp = utf8.codepoint(lastChar)
    if cp and cp >= SBASE and cp < SBASE + SCOUNT then
        local s = cp - SBASE
        local jungIdx = math.floor((s % NCOUNT) / TCOUNT)
        local jongIdx = s % TCOUNT
        
        if jongIdx == 0 and (jungIdx == 0 or jungIdx == 20) then
            return true
        end
    end
    return false
end

local function getBotSelectedWord(matchedList)
    if isFirstTurn() then
        local firstTurnAllowedEndings = {
            ["듬"] = true, ["빔"] = true, ["슴"] = true, ["듁"] = true, ["랒"] = true,
            ["른"] = true, ["쨈"] = true, ["넷"] = true, ["즙"] = true, ["뭇"] = true
        }
        
        for _, w in ipairs(matchedList) do
            local lastChar = ""
            for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
            if firstTurnAllowedEndings[lastChar] then
                return w
            end
        end
    end

    for _, w in ipairs(matchedList) do
        local lastChar = ""
        for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
        if P_TIER1[lastChar] then
            return w
        end
    end

    for _, w in ipairs(matchedList) do
        local lastChar = ""
        for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
        if P_TIER2[lastChar] then
            return w
        end
    end

    for _, w in ipairs(matchedList) do
        local lastChar = ""
        for _, cp in utf8.codes(w) do lastChar = utf8.char(cp) end
        if P_TIER3[lastChar] then
            return w
        end
    end

    local tierMatched = {}
    for _, w in ipairs(matchedList) do
        if not isExcludedEndingWord(w) then
            table.insert(tierMatched, w)
        end
    end

    if #tierMatched > 0 then
        table.sort(tierMatched, function(a, b)
            local scoreA = getPriorityScore(a)
            local scoreB = getPriorityScore(b)
            if scoreA ~= scoreB then return scoreA < scoreB end
            return utf8.len(b) < utf8.len(a)
        end)
        return tierMatched[1]
    end

    if #matchedList > 0 then
        return matchedList[1]
    end

    return nil
end

    local function isSeatOccupied(chaisePart)
        if not chaisePart then return true end
        if chaisePart:IsA("Seat") or chaisePart:IsA("VehicleSeat") then
            if chaisePart.Occupant then return true end
        end
        if chaisePart:IsA("BasePart") then
            for _, touchingPart in ipairs(chaisePart:GetTouchingParts()) do
                local character = touchingPart.Parent
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    local player = Players:GetPlayerFromCharacter(character)
                    if player and player ~= _lp then return true end
                end
            end
        end
        local chaisePos = chaisePart.Position
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= _lp and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - chaisePos).Magnitude < 2.5 then return true end
            end
        end
        return false
    end

                    local occupiedCount = 0
                    for _, playerChairFolder in ipairs(chairsFolder:GetChildren()) do
                        local chaise = playerChairFolder:FindFirstChild("chaise")
                        if chaise then
                            if isSeatOccupied(chaise) then occupiedCount = occupiedCount + 1
                            else table.insert(emptySeats, chaise) end
                        end
                    end
                    local isFourPlayer = child.Name:match("4Player") or (#chairsFolder:GetChildren() >= 4)
                    local triggerCondition = isFourPlayer and (occupiedCount >= 3) or (not isFourPlayer and occupiedCount >= 1)
                    if triggerCondition and #emptySeats > 0 then
                        local alreadyNear = false
                        for _, playerChairFolder in ipairs(chairsFolder:GetChildren()) do
                            local chaise = playerChairFolder:FindFirstChild("chaise")
                            if chaise and (rootPart.Position - chaise.Position).Magnitude < 4.0 then alreadyNear = true break end
                        end
                        if not alreadyNear then
                            local targetSeat = emptySeats[1]
                            
                            if config.autoFarmMode == "Walk" then
                                humanoid:MoveTo(targetSeat.Position)
                            else
                                rootPart.CFrame = targetSeat.CFrame + Vector3.new(0, 3, 0)
                            end

                            isTpCoolingDown = true
                            task.delay(TP_COOLDOWN_TIME, function() isTpCoolingDown = false end)
                            return true
                        end
                    end
                else
                    if #child:GetChildren() > 0 then
                        local result = scanFolder(child)
                        if result then return true end
                    end
                end
            end
            return false
        end
        scanFolder(baseFolder)
    end

    OthersTab:Toggle({
        Title = "자동 앉기 위젯 표시 (ON/OFF)",
        Desc = "켜면 화면에 자동 앉기 위젯이 나타나고, 끄면 숨겨집니다.",
        Value = false,
        Callback = function(v) 
            autoFarmGui.Enabled = v
        end
    })

    toggleBtn.Activated:Connect(function()
        isAutoTpEnabled = not isAutoTpEnabled
        if isAutoTpEnabled then
            toggleBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 50)
            toggleBtn.Text = "자동 앉기 (ON)"
        else
            toggleBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
            toggleBtn.Text = "자동 앉기 (OFF)"
            isTpCoolingDown = false 
        end
    end)

    RunService.Heartbeat:Connect(function()
        if not autoFarmGui.Enabled or not isAutoTpEnabled then return end
        pcall(function() checkAndTeleportAllTables() end)
    end)

    updateFilteredWordsView()
end

local function runLoadingSequence()
    local LoadingScreenGui = Instance.new("ScreenGui")
    LoadingScreenGui.Name = "WHITE'S_SCRIPT_BoundedTextLoadingScreen"
    LoadingScreenGui.IgnoreGuiInset = true
    LoadingScreenGui.Parent = pcall(function() return game:GetService("CoreGui") end) and game:GetService("CoreGui") or targetGuiParent

    local MainBackground = Instance.new("Frame", LoadingScreenGui)
    MainBackground.Size = UDim2.new(1, 0, 1, 0)
    MainBackground.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
    MainBackground.BackgroundTransparency = 0.15

    local CenterContainer = Instance.new("Frame", MainBackground)
    CenterContainer.AnchorPoint = Vector2.new(0.5, 0.5)
    CenterContainer.Position = UDim2.new(0.5, 0, 0.5, -20)
    CenterContainer.Size = UDim2.new(0, 450, 0, 300)
    CenterContainer.BackgroundTransparency = 1

    local TitleLabel = Instance.new("TextLabel", CenterContainer)
    TitleLabel.AnchorPoint = Vector2.new(0.5, 0)
    TitleLabel.Position = UDim2.new(0.5, 0, 0, 115)
    TitleLabel.Size = UDim2.new(1, 0, 0, 35)
    TitleLabel.BackgroundTransparency = 1
    TitleLabel.Text = "Loading Engine..."
    TitleLabel.TextColor3 = Color3.fromRGB(240, 240, 240)
    TitleLabel.Font = Enum.Font.GothamBold
    TitleLabel.TextSize = 28

    local TextProgressBar = Instance.new("TextLabel", CenterContainer)
    TextProgressBar.AnchorPoint = Vector2.new(0.5, 0)
    TextProgressBar.Position = UDim2.new(0.5, 0, 0, 170)
    TextProgressBar.Size = UDim2.new(1, 0, 0, 25)
    TextProgressBar.BackgroundTransparency = 1
    TextProgressBar.TextColor3 = Color3.fromRGB(0, 170, 255)
    TextProgressBar.Font = Enum.Font.Code
    TextProgressBar.TextSize = 18

    local StatusLabel = Instance.new("TextLabel", CenterContainer)
    StatusLabel.AnchorPoint = Vector2.new(0.5, 0)
    StatusLabel.Position = UDim2.new(0.5, 0, 0, 205)
    StatusLabel.Size = UDim2.new(1, 0, 0, 20)
    StatusLabel.BackgroundTransparency = 1
    StatusLabel.TextColor3 = Color3.fromRGB(160, 160, 160)
    StatusLabel.Font = Enum.Font.Gotham
    StatusLabel.TextSize = 14

    local totalLength = 30
    TextProgressBar.Text = "[" .. string.rep(" ", totalLength) .. "]"

    local isDataDownloaded = false
    
    task.spawn(function()
        StatusLabel.Text = "Downloading Core Dictionaries..."
        
        local rawGraph = downloadData(urls.graphUrl)
        if rawGraph then
            pcall(function()
                local decoded = HttpService:JSONDecode(rawGraph)
                if decoded then 
                    WordGraphData = decoded 
                end
            end)
        end

        local rawDueum = downloadData(urls.dueumUrl)
        if rawDueum then
            pcall(function()
                local decoded = HttpService:JSONDecode(rawDueum)
                if decoded then DueumMap = decoded end
            end)
        end

        local tempWords = {}
        local function parseAndInsert(rawText)
            if not rawText then return end
            for cleanWord in rawText:gmatch('["\'](.-)["\']') do
                cleanWord = cleanWord:match("^%s*(.-)%s*$")
                if cleanWord ~= "" and utf8.len(cleanWord) > 1 then 
                    table.insert(tempWords, cleanWord) 
                end
            end
            if #tempWords == 0 then
                for line in rawText:gmatch("[^\r\n]+") do
                    local cleaned = line:match("^%s*(.-)%s*$")
                    if cleaned then
                        cleaned = cleaned:gsub('^["\']', ''):gsub('["\']$', ''):gsub(',$', '')
                        cleaned = cleaned:match("^%s*(.-)%s*$")
                        if cleaned ~= "" and utf8.len(cleaned) > 1 then 
                            table.insert(tempWords, cleaned) 
                        end
                    end
                end
            end
        end

        StatusLabel.Text = "Downloading Opening Dict & Main DB..."
        local rawOpening = downloadData(urls.openingDictUrl)
        if rawOpening then
            pcall(function()
                local decoded = HttpService:JSONDecode(rawOpening)
                if type(decoded) == "table" then
                    for _, w in ipairs(decoded) do
                        if type(w) == "string" then table.insert(tempWords, w) end
                    end
                end
            end)
            if #tempWords == 0 then
                parseAndInsert(rawOpening)
            end
        end

        local rawMain = downloadData(urls.dictUrl)
        if rawMain then
            parseAndInsert(rawMain)
        end

        StatusLabel.Text = "Parsing & Removing Duplicates..."
        
        local rawWordsCount = #tempWords
        local chunkSize = 50 
        local lastYieldTime = os.clock()
        local temporaryMap = {}
        
        for i = 1, rawWordsCount, chunkSize do
            for j = i, math.min(i + chunkSize - 1, rawWordsCount) do
                local word = tempWords[j]
                if not temporaryMap[word] then
                    temporaryMap[word] = true
                    table.insert(WordDB, word)
                    MainDictMap[word] = true
                end
            end
            
            if os.clock() - lastYieldTime > 0.1 then
                task.wait()
                lastYieldTime = os.clock()
            end
        end
        temporaryMap = nil
        
        isDataDownloaded = true
    end)

    task.spawn(function()
        local currentFilled = 0
        for i = 1, totalLength do
            currentFilled = i
            local filledPart = string.rep("|", currentFilled)
            local emptyPart = string.rep(" ", totalLength - currentFilled)
            TextProgressBar.Text = "[" .. filledPart .. emptyPart .. "]"
            StatusLabel.Text = "Injecting Automata Data... " .. math.floor((i / totalLength) * 100) .. "%"
            task.wait(0.03) 
        end
        
        StatusLabel.Text = "Finalizing Graphics Engine..."
        while not isDataDownloaded do
            task.wait(0.02)
        end
        
        task.spawn(function()
            local success, err = pcall(initializeWindUIWindow)
            if not success then
                warn("UI 로드 에러 발생: " + tostring(err))
            end
        end)
        
        task.spawn(function()
            pcall(function()
                table.sort(WordDB)
            end)
        end)
        
        task.wait(0.1)
        
        -- [오류 수정 부분]: 로딩 화면이 마우스 및 모바일 터치 입력을 계속 가로채지 못하도록 닿는 모든 요소에 Active = false 및 터치 이벤트 해제 처리 추가
        MainBackground.Active = false
        CenterContainer.Active = false

        TweenService:Create(MainBackground, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
        TweenService:Create(TitleLabel, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        TweenService:Create(TextProgressBar, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        TweenService:Create(StatusLabel, TweenInfo.new(0.2), {TextTransparency = 1}):Play()
        
        task.wait(0.3)
        LoadingScreenGui:Destroy()
        
        collectgarbage("collect")
    end)
end

task.spawn(function()
    if checkWhitelistValid() then
        runLoadingSequence()
    end
end)
